#include <LiquidCrystal.h>
const int rs = 12, en = 11, d4 = 5, d5 = 4, d6 = 3, d7 = 2;
LiquidCrystal lcd(rs, en, d4, d5, d6, d7);
volatile int ore = 12;
volatile int minute = 30;
volatile int secunde = 0;
volatile bool secundaScursa = false;
void setup()
{  
   lcd.begin(16, 2);
   Serial.begin(9600);
   cli();                 
   TCCR1A = 0;             
   TCCR1B = 0;             
   TCNT1  = 0;
   OCR1A = 62499;
   TCCR1B |= (1 << WGM12);
   TCCR1B |= (1 << CS12);
   TIMSK1 |= (1 << OCIE1A);
   sei();
}
ISR(TIMER1_COMPA_vect) {
  secunde++;
  if (secunde >= 60) {
    secunde = 0;
    minute++;
    if (minute >= 60) {
      minute = 0;
      ore++;
      if (ore >= 24) {
        ore = 0;
      }
    }
  }
  secundaScursa = true;
}

void loop() {
  if (secundaScursa) {
    secundaScursa = false; 
    int valBruta = analogRead(A0);
  	float tensiune = valBruta * (5.0 / 1023.0);
  	float temperatura = (tensiune - 0.5) * 100.0;
    char buffTimp[9];
    sprintf(buffTimp, "%02d:%02d:%02d", ore, minute, secunde);
    Serial.print("Ora: ");
    Serial.print(buffTimp);
    Serial.print(" | Temp: ");
    Serial.print(temperatura, 1);
    Serial.println(" C");
    lcd.setCursor(0, 0);
    lcd.print("Ora: ");
    lcd.print(buffTimp);

    lcd.setCursor(0, 1);
    lcd.print("Temp: ");
    lcd.print(temperatura);
    lcd.print("°C");
  }
}
